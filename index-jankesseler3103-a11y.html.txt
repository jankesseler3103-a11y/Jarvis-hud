  function api(messages,signal){
    var defs=toolList.map(function(t){return{name:t.name,description:t.description,input_schema:t.schema}});
    return fetch('https://api.anthropic.com/v1/messages',{
      method:'POST',signal:signal,
      headers:{'content-type':'application/json','x-api-key':apiKey,'anthropic-version':'2023-06-01','anthropic-dangerous-direct-browser-access':'true'},
      body:JSON.stringify({model:MODEL,max_tokens:800,system:context(),tools:defs,messages:messages})
    }).then(function(r){
      return r.json().catch(function(){return{}}).then(function(j){
        if(!r.ok){var e=new Error((j.error&&j.error.message)||'API-Fehler');e.status=r.status;throw e}
        return j;
      });
    });
  }
  function textOf(j){return j.content.filter(function(c){return c.type==='text'}).map(function(c){return c.text}).join(' ').trim()}
  function loop(messages,signal,depth){
    return api(messages,signal).then(function(j){
      var calls=j.content.filter(function(c){return c.type==='tool_use'});
      if(j.stop_reason==='tool_use'&&calls.length&&depth<4){
        var results=calls.map(function(c){
          var out,err=false;
          try{var tl=toolList.filter(function(t){return t.name===c.name})[0];if(!tl)throw new Error('Unbekanntes Werkzeug');out=String(tl.run(c.input||{}))}
          catch(e){out='Error: '+e.message;err=true}
          return{type:'tool_result',tool_use_id:c.id,content:out,is_error:err};
        });
        var next=messages.concat([{role:'assistant',content:j.content},{role:'user',content:results}]);
        var pre=textOf(j);
        return loop(next,signal,depth+1).then(function(t){return(pre?pre+' ':'')+t});
      }
      return textOf(j);
    });
  }

  function ask(t){
    if(!apiKey){say('Dafür brauche ich Zugang zu meinem Denkapparat, Sir. Tragen Sie unter „KI-Schlüssel“ einen API-Schlüssel ein.',true);return}
    if(busy){say('Einen Moment, Sir, ich denke noch.',true);return}
    busy=true;toolList=tools();
    var li=document.createElement('li');li.className='jv';
    var w=document.createElement('span');w.className='who';w.textContent='J.A.R.V.I.S.';
    var p=document.createElement('span');p.textContent='Ich denke nach …';
    li.appendChild(w);li.appendChild(p);logEl.insertBefore(li,logEl.firstChild);
    while(logEl.children.length>5)logEl.removeChild(logEl.lastChild);
    turns.push({role:'user',content:t});
    if(turns.length>12)turns=turns.slice(-12);
    while(turns.length&&turns[0].role!=='user')turns.shift();
    ctl=new AbortController();
    loop(turns.slice(),ctl.signal,0).then(function(txt){
      txt=txt||'Erledigt, Sir.';
      p.textContent=txt;turns.push({role:'assistant',content:txt});speak(txt);
    }).catch(function(e){
      turns.pop();
      var msg;
      if(e&&e.name==='AbortError')msg='Abgebrochen.';
      else if(e&&(e.status===401||e.status===403))msg='Der Schlüssel wurde abgelehnt, Sir. Bitte prüfen Sie ihn unter „KI-Schlüssel“.';
      else if(e&&e.status===429)msg='Das Limit ist erreicht, Sir. Bitte später erneut.';
      else if(e&&e.status)msg='Der Dienst meldet einen Fehler: '+e.message;
      else msg='Keine Verbindung zu meinem Denkapparat, Sir. Sind Sie online?';
      li.className='jv warn';p.textContent=msg;
    }).then(function(){busy=false;ctl=null});
  }
